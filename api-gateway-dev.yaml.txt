server:
  port: 8080

spring:
  application:
    name: api-gateway

  profiles:
    active: dev

  security:
    oauth2:
      resourceserver:
        jwt:
          issuer-uri: http://localhost:9000

  data:
    redis:
      host: localhost
      port: 6379

  cloud:
    gateway:
      default-filters:
        - name: RequestRateLimiter
          args:
            redis-rate-limiter.replenishRate: 5
            redis-rate-limiter.burstCapacity: 10

      routes:
        # Account Service
        - id: account-service
          uri: lb://ACCOUNT-SERVICE
          predicates:
            - Path=/api/account/**
          filters:
            - StripPrefix=1

        # Transaction Service
        - id: transaction-service
          uri: lb://TRANSACTION-SERVICE
          predicates:
            - Path=/api/transaction/**
          filters:
            - StripPrefix=1
            - name: CircuitBreaker
              args:
                name: transactionCB

        # Auth Service - Custom authentication APIs
        - id: auth-service
          uri: lb://AUTH-SERVICE
          predicates:
            - Path=/api/auth/**
          filters:
            - StripPrefix=1

        # Auth Service - OAuth2 endpoints
        - id: oauth2-service
          uri: lb://AUTH-SERVICE
          predicates:
            - Path=/oauth2/**

eureka:
  client:
    service-url:
      defaultZone: http://localhost:8761/eureka/

management:
  endpoints:
    web:
      exposure:
        include:
          - health
          - info
          - prometheus
